最小骨架

// NuGet: Microsoft.Extensions.Hosting（一个包带全 DI / Configuration / Options / Logging）
var builder = Host.CreateApplicationBuilder(args);

builder.Configuration.AddJsonFile("extra.json", optional: true);   // 追加配置源
builder.Services.AddSingleton<MyService>();                        // DI
builder.Services.AddHostedService<MyWorker>();                     // 后台任务
builder.Logging.AddConsole();                                      // 日志（默认已有）

using IHost host = builder.Build();

// 长驻：
await host.RunAsync();
// 或 跑一次就退：
// var svc = host.Services.GetRequiredService<MyService>(); svc.Run(); return 0;

ContentRoot   = 当前工作目录（当前目录是 C:\Windows\System32 时退回 exe 目录）

Host 配置     ← DOTNET_* 环境变量、命令行
               （DOTNET_ENVIRONMENT / --environment 决定环境名）

应用配置      ← 顺序即优先级，最后胜出
   appsettings.json
   appsettings.{Environment}.json
   User Secrets（仅 Development）
   环境变量（无前缀）
   命令行

日志 Provider ← Console、Debug、EventSource、EventLog（仅 Windows）

DI 容器校验   ← 仅 Development：ValidateScopes = true、ValidateOnBuild = true

IHostedService 与 BackgroundService

// 启动时做一件事就返回（预热、注册）
public sealed class Warmup : IHostedService
{
    public Task StartAsync(CancellationToken ct) { /* 被 Host await，必须返回 */ return Task.CompletedTask; }
    public Task StopAsync(CancellationToken ct)  => Task.CompletedTask;
}

// 长循环
public sealed class Worker(ILogger<Worker> logger, IServiceScopeFactory scopeFactory) : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        await Task.Yield();                                  // 避免同步段阻塞 Host 启动
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                using var scope = scopeFactory.CreateScope();  // Singleton 里用 Scoped：每次迭代一个 Scope
                var job = scope.ServiceProvider.GetRequiredService<IJob>();
                await job.RunAsync(stoppingToken);
            }
            catch (OperationCanceledException) when (stoppingToken.IsCancellationRequested) { break; }
            catch (Exception ex)
            {
                logger.LogError(ex, "Iteration failed");     // 可恢复错误：自己兜住
            }
            await Task.Delay(TimeSpan.FromSeconds(30), stoppingToken);   // token 一定要传
        }
    }
}

builder.Services.AddHostedService<Warmup>();   // 注册顺序 = 启动顺序
builder.Services.AddHostedService<Worker>();   // 关闭顺序 = 倒序

生命周期顺序

启动  IHostLifetime.WaitForStartAsync
      IStartupValidator（ValidateOnStart）
      每个服务（注册顺序）：StartingAsync → StartAsync → StartedAsync
      ApplicationStarted

关闭  ApplicationStopping                 ← "别接新活"
      每个服务（倒序）：StoppingAsync → StopAsync → StoppedAsync   ← "收尾"，总预算 ShutdownTimeout
      ApplicationStopped
      IHostLifetime.StopAsync

HostOptions

属性默认appsettings 键说明
ShutdownTimeout30 sshutdownTimeoutSeconds超时后 token 取消、不再等待。与 docker stop / K8s grace period 一起调
StartupTimeout无限startupTimeoutSeconds生产建议设置，防止启动卡死"假活"
ServicesStartConcurrentlyfalseservicesStartConcurrentlytrue = 并行启动，失去顺序保证
ServicesStopConcurrentlyfalseservicesStopConcurrently同上
BackgroundServiceExceptionBehaviorStopHost—.NET 6+：未捕获异常 → 记日志 → 停 Host。Ignore = .NET 5 的静默吞掉
builder.Services.Configure<HostOptions>(o =>
{
    o.ShutdownTimeout = TimeSpan.FromSeconds(60);
    o.StartupTimeout  = TimeSpan.FromSeconds(30);
});

优雅退出

方式走关闭流程？说明
Ctrl+C / SIGINT、SIGTERM（docker stop）、SIGQUIT✅ConsoleLifetime 捕获
IHostApplicationLifetime.StopApplication()✅代码里主动退出的正确方式
Environment.Exit()❌StopAsync、finally 都不执行。.NET 6+ 的 Host 不再介入
SIGKILL、崩溃❌Host 无能为力
