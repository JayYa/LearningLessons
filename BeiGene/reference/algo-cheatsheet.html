N = 数据总页数，B = 可用缓冲页数。

4 · 堆向速记：守门员

堆 = 只有 K 个座位的贵宾席。堆顶 = 席位里最弱的那个 = 守门员 = 门槛。
新人比守门员强 → 踢掉守门员进来；比守门员弱 → 不让进。

求最大 K 个 → 小顶堆　　求最小 K 个 → 大顶堆

数组堆：父 (i-1)/2，子 2i+1 / 2i+2。Floyd 自底向上建堆是 O(n)。

5 · 四段代码骨架（默写用）

5.1 快排 + 分区（分区被 5.2 复用）

public static void QuickSort(int[] a) => Sort(a, 0, a.Length - 1);

static void Sort(int[] a, int lo, int hi) {
    while (lo < hi) {
        if (hi - lo < 16) { InsertionSort(a, lo, hi); return; }   // 16 阈值
        int p = Partition(a, lo, hi);
        if (p - lo < hi - p) { Sort(a, lo, p - 1); lo = p + 1; }  // 只递归短边
        else                 { Sort(a, p + 1, hi); hi = p - 1; }  // 栈深 O(log n)
    }
}

public static int Partition(int[] a, int lo, int hi) {
    int mid = lo + (hi - lo) / 2;                    // 防 int 溢出
    if (a[lo]  > a[mid]) Swap(a, lo, mid);           // 三数取中
    if (a[lo]  > a[hi])  Swap(a, lo, hi);
    if (a[mid] > a[hi])  Swap(a, mid, hi);
    Swap(a, mid, hi);                                // 中位数挪到 hi 当 pivot
    int pivot = a[hi], i = lo - 1;
    for (int j = lo; j < hi; j++)
        if (a[j] <= pivot) Swap(a, ++i, j);
    Swap(a, ++i, hi);
    return i;                                        // = pivot 的最终下标
}

5.2 快速选择 / 中位数

public static int QuickSelect(int[] a, int k) {      // k 为 0-based；会打乱 a
    int lo = 0, hi = a.Length - 1;
    while (lo < hi) {
        int p = Partition(a, lo, hi);
        if (p == k) return a[p];
        if (p < k) lo = p + 1; else hi = p - 1;       // 只走一边
    }
    return a[lo];
}

public static double Median(int[] a) {
    int n = a.Length;
    if (n == 0) throw new ArgumentException("空数组没有中位数", nameof(a));
    if ((n & 1) == 1) return QuickSelect(a, n / 2);
    int hiHalf = QuickSelect(a, n / 2);              // 偶数：两次选择
    int loHalf = QuickSelect(a, n / 2 - 1);          // 第二次更快（已部分划分）
    return (loHalf + hiHalf) / 2.0;
}

5.3 Top-K（最大的 K 个 → 小顶堆）

public static List<int> LargestK(IEnumerable<int> source, int k) {
    if (k <= 0) return new List<int>();
    var heap = new PriorityQueue<int, int>(k);        // 默认小顶堆
    foreach (int x in source) {                       // source 可为 File.ReadLines(...)
        if (heap.Count < k)      heap.Enqueue(x, x);
        else if (x > heap.Peek()) heap.EnqueueDequeue(x, x);   // 换守门员
    }
    var r = new List<int>(heap.Count);
    while (heap.Count > 0) r.Add(heap.Dequeue());
    r.Reverse();                                      // 小→大 反转成 大→小
    return r;
}

5.4 外部归并排序（从大到小）

static readonly IComparer<long> Desc = Comparer<long>.Create((x, y) => y.CompareTo(x));

public static void SortDescending(string inPath, string outPath, int chunkSize, string tmp) {
    Directory.CreateDirectory(tmp);
    var runs = new List<string>();
    using (var rd = new StreamReader(inPath)) {                 // 逐行读！
        var buf = new List<long>(chunkSize);
        string? line;
        while ((line = rd.ReadLine()) is not null) {
            if (line.Length == 0) continue;
            buf.Add(long.Parse(line));
            if (buf.Count == chunkSize) runs.Add(Flush(buf, tmp, runs.Count));
        }
        if (buf.Count > 0) runs.Add(Flush(buf, tmp, runs.Count));   // ★最后一块，最易漏
    }
    Merge(runs, outPath);
    foreach (var r in runs) File.Delete(r);
}

static string Flush(List<long> buf, string tmp, int i) {
    buf.Sort(Desc);
    string p = Path.Combine(tmp, $"run-{i:D4}.txt");
    using (var w = new StreamWriter(p)) foreach (long v in buf) w.WriteLine(v);
    buf.Clear();
    return p;
}

static void Merge(List<string> runs, string outPath) {
    var rs = runs.Select(p => new StreamReader(p)).ToList();
    try {
        var heap = new PriorityQueue<(long V, int R), long>(runs.Count, Desc);  // 大顶堆
        for (int i = 0; i < rs.Count; i++)
            if (TryNext(rs[i], out long v)) heap.Enqueue((v, i), v);
        using var w = new StreamWriter(outPath);
        while (heap.TryDequeue(out var top, out _)) {
            w.WriteLine(top.V);
            if (TryNext(rs[top.R], out long nx)) heap.Enqueue((nx, top.R), nx);
        }
    } finally { foreach (var r in rs) r.Dispose(); }
}

6 · 白板检查清单

中点lo + (hi - lo) / 2，不是 (lo + hi) / 2
区间全程闭区间 [lo, hi]——一致比正确更重要
k 基准先声明 0-based；中位数偶数取 n/2-1 与 n/2
堆向最大 K 个 → 小顶堆（念「守门员」）
最后一块Pass 0 循环结束后必须再 flush 一次残留
逐行读绝不 File.ReadAllLines——那否定了题目前提
释放流using / try-finally，几十个 Reader 别泄漏
空 / 单元素n=0 时 hi=-1，循环不进；中位数显式抛异常

7 · 接数据库题（PostgreSQL）

看到什么含义怎么办
Sort Method: quicksort  Memory: xxx排序在内存里完成正常
Sort Method: external merge  Disk: xxx超出 work_mem，落盘走外部归并见右列三步
Sort Method: top-N heapsortORDER BY … LIMIT n 触发的堆式 Top-N正是档位 C 在数据库里的实现

三步排查（话术顺序）：

  1. 真的需要全量排序吗？多半是分页 → keyset 分页 + 排序列索引，排序成本归零（见 索引速查表）
  2. 调大会话级 work_mem（默认仅 4MB），减少归并趟数。work_mem 是每操作、每会话各算一份 → 用 SET LOCAL，别改全局
  3. 常态需求就建索引：排序列上的 B-tree → 按索引顺序吐数据，Sort 节点整个消失

8 · 一句话话术卡

被问第一句就说
现场写排序「生产我用 Array.Sort，它是内省排序、不稳定；手写我写带三数取中和尾递归消除的快排。」
找中位数「不排序。我只要第 n/2 个，用快速选择，每次只递归一边，期望 O(n)。」
海量数据找 Top-K「容量 K 的小顶堆流式扫一遍，O(N log K)，内存只占 O(K)。」
超内存全序「外部归并排序：分块内排落盘成 run，再用堆做多路归并，内存只放段数个元素。」
数据库无索引排序「不往应用里捞。数据库自己就走外部归并，EXPLAIN 里看 Sort Method；先看要不要全量，再调 work_mem，常态就建索引。」
